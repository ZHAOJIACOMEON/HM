## 第1题：路由定义
某健身 App 需要一个欢迎接口：访问路径 `/fitness/welcome` 时，返回 JSON 数据 `{"msg": "欢迎来到智能健身平台"}`。请用 FastAPI 实现该路由。
```python

ASYNC_DATABASE_URL = 'mysql+aiomysql://root:@localhost:3306/shop?charset=utf8'
async_engine = create_async_engine(ASYNC_DATABASE_URL,
                                   echo=True,  # 输出日志
                                   pool_size=10,  # 设置连接池
                                   max_overflow=20,  # 设置连接池允许的最大连接数
                )
# 基类
class Base(DeclarativeBase):
    create_time: Mapped[datetime] = mapped_column(
                                                  DateTime,
                                                  insert_default=func.now(),
                                                  default=datetime.now,
                                                  comment="创建时间"
                                    )
    update_time: Mapped[datetime] = mapped_column(
                                                  DateTime,
                                                  onupdate=func.now(),
                                                  default=datetime.now(),
                                                  comment="修改时间"
                                    )
# 建表模型类
class Goods(Base):
    __tablename__ = 'goods'
    id: Mapped[int] = mapped_column(primary_key=True, autoincrement=True)
    name: Mapped[str] = mapped_column(String(50))
    category: Mapped[str] = mapped_column(String(50))
    price: Mapped[float] = mapped_column(Float)
    stock: Mapped[int] = mapped_column(INT)
# 建表
# 会话工厂
AsyncSessionLocal = async_sessionmaker(
  bind=async_engine,  # 绑定数据库引擎
  class_=AsyncSession,  # 指定会话类
  expire_on_commit=False  # 会话对象不过期，不重新查询数据库
)
# 依赖注入
async def get_database():
  async with AsyncSessionLocal() as session:
    try:
      yield session  # 返回数据库会话给路由处理函数
      await session.commit()  # 无异常，提交事务
    except Exception:
      await session.rollback()  # 有异常则回滚
      raise
    finally:
      await session.close()  # 关闭会话

@asynccontextmanager
async def lifespan(app: FastAPI):
    async with async_engine.begin() as conn:
            await conn.run_sync(Base.metadata.create_all)
            yield

app = FastAPI(lifespan=lifespan, )
# 路由使用
@app.get('/fitness/welcome')
async def fitness_welcome(
        db: Annotated[AsyncSession, Depends(get_database)]
    ):
    return {"msg": "欢迎来到智能健身平台"}
```

## 第16题：中间件编程
使用 `@app.middleware("http")` 实现一个**请求日志中间件**：记录每个请求的方法（`request.method`）和路径（`request.url.path`），并将日志信息添加到响应头 `X-Request-Log` 中返回给客户端。
```python
from fastapi import FastAPI, Request

app = FastAPI()

@app.middleware('http')
async def middle_ware(request, call_next):
    # 1. 记录方法与路径
    log_msg = f"{request.method} {request.url.path}"

    # 2. 继续执行后续处理，拿到响应
    response = await call_next(request)

    # 3. 把日志写进响应头
    response.headers["X-Request-Log"] = log_msg

    return response
```

## 第23题：条件查询 - 与非查询
定义两个接口：
- 接口1 `GET /pets/and`：查询**年龄大于 1 且价格小于 500** 的宠物，使用 `&`（与）组合
- 接口2 `GET /pets/or`：查询**年龄大于等于 3 或品种为"猫"** 的宠物，使用 `|`（或）组合
```python
from fastapi import FastAPI, Depends, HTTPException
from typing import Annotated
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy import select
from pydantic import BaseModel

app = FastAPI()

async def get_session():
    async with async_session() as session:
        yield session

@app.get('/pets/and')
async def pets_and(
        age: int,
        price: float,
        db: Annotated[AsyncSession, Depends(get_session)]
    ):
    stmt = select(Pet).where((Pet.age > age), (Pet.price > price))
    result = await db.execute(stmt)
    pets = result.scalars().all()

    return pets

@app.get('/pets/or')
async def pets_or(
        age: int,
        species: str,
        db: Annotated[AsyncSession, Depends(get_session)]
    ):
    stmt = select(Pet).where((Pet.age >= age) | (Pet.species = species))
    result = await db.execute(stmt)
    pets = result.scalars().all()
```

## 第28题：修改数据
定义 Pydantic 模型 `PetUpdate`（name、price），接口 `PUT /pets/{pet_id}`。先通过 `select().where()` 查询宠物，再对 name 和 price 重新赋值，最后 `commit()` 提交。宠物不存在抛 404。
```python
from fastapi import FastAPI, Depends, HTTPException
from typing import Annotated
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy import select
from pydantic import BaseModel

app = FastAPI()

async def get_session():
    async with async_session() as session:
        yield session

class PetUpdate(BaseModel):
    name: str
    price: float

@app.post('/pets/{pet_id}')
async def pets_update(
        pet_id: int,
        pet_data: PetUpdate,
        db: Annotated[AsyncSession, Depends(get_session)]
    ):
    result = await db.execute(select(Pet).where(Pet.id == pet_id))
    pet = result.scalars().first()
    if not pet:
        raise HTTPException(status_code=404, detail="宠物不存在")
    # 2. 重新赋值
    pet.name = pet_data.name
    pet.price = pet_data.price
    # 3. 提交
    await db.commit()
    return {"msg": "修改成功"}
```

## 第30题：综合 - 新增前查重
定义接口 `POST /pets/check`，接收 `PetCreate` 请求体。新增前先使用 `select().where()` 查询是否已存在**同名同品种**的宠物：
- 如果已存在，抛出 `HTTPException`（400，detail `"宠物已存在，请勿重复添加"）
- 如果不存在，执行新增流程（`model_dump()` → `add()` → `commit()`），返回 `{"msg": "新增成功", "id": pet.id}`
```python
from fastapi import FastAPI, Depends, HTTPException
from typing import Annotated
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy import select
from pydantic import BaseModel

async def get_session():
    async with async_session() as session:
        yield session

    species: str
    age: int
    price: float

        pet_data: PetCreate,
        db: Annotated[AsyncSession, Depends(get_session)]
    ):
    result = await db.execute(select(Pet).where(Pet.name == pet_data.name, Pet.species == pet_data.species))
    pet = result.scalars().first()
    if pet:
        raise HTTPException(status_code=404, detail="该宠物已存在，请勿重复添加~")

    await db.commit()

    return {
        "msg": "新增成功", 
        "id": pet.id
    }
```
